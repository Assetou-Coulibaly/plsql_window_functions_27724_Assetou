1.Ranking Functions 
                    SELECT c.customer_id,
                           c.customer_name,
                          SUM(t.amount) AS total_amount,
                          RANK() OVER (ORDER BY SUM(t.amount) DESC) AS rank_total
                    FROM Customers c 
                    JOIN Transactions t ON c.customer_id = t.customer_id
                    GROUP BY c.customer_id, c.customer_name
                    ORDER BY rank_total;
2.Aggregate Window Functions

                            SELECT t.transaction_id,
                                   t.transaction_date,
                                   t.amount,
                                  SUM(t.amount) OVER (ORDER BY t.transaction_date) AS running_total
                             FROM Transactions t
                            ORDER BY t.transaction_date;
3.Navigation Functions

                        SELECT t.transaction_id,
                               t.transaction_date,
                               t.amount,
                              LAG(t.amount,1) OVER (ORDER BY t.transaction_date) AS previous_amount,
                              LEAD(t.amount,1) OVER (ORDER BY t.transaction_date) AS next_amount
                         FROM Transactions t
                         ORDER BY t.transaction_date;
4.Distribution Functions

                       SELECT c.customer_id,
                              c.customer_name,
                              SUM(t.amount) AS total_amount,
                              NTILE(4) OVER (ORDER BY SUM(t.amount) DESC) AS quartile 
                        FROM Customers c
                        JOIN Transactions t ON c.customer_id = t.customer_id
                        GROUP BY c.customer_id, c.customer_name
                        ORDER BY quartile;
